  (function() {
    // fixed credentials as requested: name = "monstar", password = "monster12"
    const VALID_USER = "monstar";
    const VALID_PASS = "monster12";

    // DOM elements
    const loginNameInput = document.getElementById('loginName');
    const loginPassInput = document.getElementById('loginPass');
    const accessBtn = document.getElementById('accessBtn');
    const accessMsg = document.getElementById('accessMsg');
    const prefillBtn = document.getElementById('prefillCreds');

    // helper to set message (with icon)
    function setMessage(text, isSuccess = false) {
      accessMsg.innerHTML = isSuccess ? 
        `<i class="fas fa-check-circle" style="color:#2ce57a;"></i> ${text}` : 
        `<i class="fas fa-exclamation-triangle" style="color:#ffb86b;"></i> ${text}`;
    }

    // check credentials and show appropriate message
    function checkAndDisplay() {
      const enteredUser = loginNameInput.value.trim();
      const enteredPass = loginPassInput.value;  // don't trim password (exact)

      if (enteredUser === VALID_USER && enteredPass === VALID_PASS) {
        setMessage(`Access GRANTED · welcome back, monstar · web root unlocked`, true);
        // optional: make it extra flashy
        accessMsg.style.color = "#b0ffc8";
      } else {
        setMessage(`access DENIED · who are you? valid credentials: monstar / monster12`);
        accessMsg.style.color = "#ffb86b";
      }
    }

    // event listener for access button
    accessBtn.addEventListener('click', function(e) {
      e.preventDefault();  // avoid any form-like behavior
      checkAndDisplay();
    });

    // prefill / reset credentials to the correct ones (monstar / monster12)
    function resetToValid() {
      loginNameInput.value = VALID_USER;
      loginPassInput.value = VALID_PASS;
      setMessage(`credentials restored to monstar / monster12 · try now`, false);
    }

    prefillBtn.addEventListener('click', function(e) {
      e.preventDefault();
      resetToValid();
    });

    // also trigger check if user presses enter in any field (optional)
    [loginNameInput, loginPassInput].forEach(input => {
      input.addEventListener('keypress', function(e) {
        if (e.key === 'Enter') {
          e.preventDefault();
          checkAndDisplay();
        }
      });
    });

    // initial message on page load (empty or show hint)
    setMessage(`enter name/pass and click "ACCESS GRANT"`);

    // but we pre-filled with correct values? yes inputs already set to monstar/monster12
    // optionally set a welcome because we prefilled — but let user decide to click.
